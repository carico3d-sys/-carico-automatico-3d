# 1. Installare Docker (una volta sola)
curl -fsSL https://get.docker.com | sh

# 2. Clonare il progetto da GitHub

cd -carico-automatico-3d

# 4. Avviare lo stack Compose: PostgreSQL resta nella rete interna Docker

docker compose exec web python manage.py createsuperuser
